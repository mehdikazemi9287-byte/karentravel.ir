  // Countdown Timer Simulation
  (function () {
    var totalSeconds = 8 * 60 + 16;
    var timerEl = document.getElementById('countdown-timer');
    if (!timerEl) return;

    var toPersianDigits = function (str) {
      var id = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
      return str.replace(/[0-9]/g, function (w) {
        return id[+w];
      });
    };

    var interval = setInterval(function () {
      if (totalSeconds <= 0) {
        clearInterval(interval);
        timerEl.textContent = '۰۰:۰۰';
        return;
      }
      totalSeconds--;
      var mins = Math.floor(totalSeconds / 60);
      var secs = totalSeconds % 60;
      var formatted = (mins < 10 ? '0' : '') + mins + ':' + (secs < 10 ? '0' : '') + secs;
      timerEl.textContent = toPersianDigits(formatted);
    }, 1000);
  })();

  // Collapsible Passenger Information
  (function () {
    var btn = document.getElementById('passenger-toggle-btn');
    var content = document.getElementById('passenger-content');
    var chevron = document.getElementById('passenger-chevron');
    if (!btn || !content || !chevron) return;

    var isExpanded = true;
    btn.addEventListener('click', function () {
      isExpanded = !isExpanded;
      if (isExpanded) {
        content.classList.remove('hidden');
        chevron.style.transform = 'rotate(0deg)';
      } else {
        content.classList.add('hidden');
        chevron.style.transform = 'rotate(180deg)';
      }
    });
  })();

  (function () {
    var container = document.getElementById('payment-options-container');
    if (!container) return;
    var radios = container.querySelectorAll('input[type="radio"]');

    function updateRadioIcons() {
      radios.forEach(function (radio) {
        var label = radio.closest('.pay-option');
        if (!label) return;
        var icon = label.querySelector('.pay-radio-icon');
        if (!icon) return;
        if (radio.checked) {
          icon.classList.remove('opacity-0');
        } else {
          icon.classList.add('opacity-0');
        }
      });
    }

    radios.forEach(function (radio) {
      radio.addEventListener('change', updateRadioIcons);
    });
    updateRadioIcons();
  })();

<script>
    function showToast(message) {
      const toast = document.getElementById('appToast');
      if (!toast) return;
      toast.innerText = message;
      toast.classList.remove('opacity-0', '-translate-y-2', 'pointer-events-none');
      toast.classList.add('opacity-100', 'translate-y-0');
      setTimeout(() => {
        toast.classList.add('opacity-0', '-translate-y-2', 'pointer-events-none');
        toast.classList.remove('opacity-100', 'translate-y-0');
      }, 2500);
    }

    function simulatePdfDownload() {
      showToast('در حال آماده‌سازی و دانلود فایل PDF بلیت...');
      setTimeout(() => {
        showToast('فایل بلیت الکترونیک بارگیری شد ✓');
      }, 1500);
    }

    function addToCalendar() {
      showToast('رویداد پرواز به تقویم گوگل شما اضافه گردید');
    }

    function toggleRefundDetails() {
      const details = document.getElementById('refundDetails');
      const chevron = document.getElementById('refundChevron');
      if (!details || !chevron) return;
      const isHidden = details.classList.contains('hidden');
      if (isHidden) {
        details.classList.remove('hidden');
        chevron.classList.add('rotate-180');
      } else {
        details.classList.add('hidden');
        chevron.classList.remove('rotate-180');
      }
    }

    function toggleRefundDrawer() {
      toggleRefundDetails();
      const details = document.getElementById('refundDetails');
      if (details) {
        details.scrollIntoView({ behavior: 'smooth', block: 'center' });
      }
    }

    function requestRefundModal() {
      if (confirm('آیا از ارسال درخواست استعلام کنسلی و استرداد بلیت اطمینان دارید؟')) {
        showToast('درخواست کنسلی برای پشتیبان (مریم رحمانی) ارسال شد');
      }
    }
