<script>
  let currentMode = 'partial';
  const TICKET_PRICE = 17250000;
  const PENALTY_RATE = 0.40;
  const BONUS_RATE = 0.02;

  function switchCancelMode(mode) {
    currentMode = mode;
    const tabPartial = document.getElementById('tab-partial');
    const tabAll = document.getElementById('tab-all');
    const p1 = document.getElementById('passenger-1');
    const p2 = document.getElementById('passenger-2');

    if (mode === 'all') {
      tabAll.className = 'py-2 rounded-xl font-label-lg text-label-lg transition-all text-on-surface bg-surface-container-lowest shadow-sm flex items-center justify-center gap-1.5';
      tabPartial.className = 'py-2 rounded-xl font-label-lg text-label-lg transition-all text-on-surface-variant hover:text-on-surface flex items-center justify-center gap-1.5';
      p1.checked = true;
      p2.checked = true;
      p1.disabled = true;
      p2.disabled = true;
    } else {
      tabPartial.className = 'py-2 rounded-xl font-label-lg text-label-lg transition-all text-on-surface bg-surface-container-lowest shadow-sm flex items-center justify-center gap-1.5';
      tabAll.className = 'py-2 rounded-xl font-label-lg text-label-lg transition-all text-on-surface-variant hover:text-on-surface flex items-center justify-center gap-1.5';
      p1.disabled = false;
      p2.disabled = false;
      p1.checked = true;
      p2.checked = false;
    }
    updateCalculations();
  }

  function selectReason(button, reasonKey) {
    const buttons = document.querySelectorAll('.reason-pill');
    buttons.forEach(btn => {
      btn.className = 'reason-pill p-3 rounded-2xl text-right bg-surface-container-low text-on-surface hover:bg-surface-container transition-all flex flex-col justify-between h-20';
      const icon = btn.querySelector('.material-symbols-outlined');
      if (icon) icon.className = 'material-symbols-outlined text-secondary text-[20px]';
    });

    button.className = 'reason-pill p-3 rounded-2xl text-right bg-secondary text-on-secondary transition-all flex flex-col justify-between h-20 shadow-sm';
    const activeIcon = button.querySelector('.material-symbols-outlined');
    if (activeIcon) activeIcon.className = 'material-symbols-outlined text-on-secondary text-[20px]';

    const medicalBox = document.getElementById('medical-attachment');
    if (reasonKey === 'medical') {
      medicalBox.classList.remove('hidden');
    } else {
      medicalBox.classList.add('hidden');
    }
  }

  function updatePayoutMethod() {
    updateCalculations();
  }

  function updateCalculations() {
    const p1 = document.getElementById('passenger-1');
    const p2 = document.getElementById('passenger-2');

    let count = 0;
    if (p1.checked) count++;
    if (p2.checked) count++;

    const totalFare = count * TICKET_PRICE;
    const penaltyAmount = totalFare * PENALTY_RATE;
    const baseRefund = totalFare - penaltyAmount;
    const walletWithBonus = baseRefund + (baseRefund * BONUS_RATE);

    const summaryRefundEl = document.getElementById('summary-refund');
    const summaryPenaltyEl = document.getElementById('summary-penalty');
    const walletAmountEl = document.getElementById('wallet-amount');
