  // 360 Panorama Drag Micro-Interaction
  const container = document.getElementById('panorama-container');
  const panoBg = document.getElementById('pano-bg');
  const angleDisplay = document.getElementById('angle-display');
  let isDragging = false;
  let startX = 0;
  let currentOffset = -20; // in percent

  if (container && panoBg) {
    const handleTouchStart = (e) => {
      isDragging = true;
      startX = e.touches ? e.touches[0].clientX : e.clientX;
    };

    const handleTouchMove = (e) => {
      if (!isDragging) return;
      const x = e.touches ? e.touches[0].clientX : e.clientX;
      const deltaX = (x - startX) * 0.15;
      let newOffset = currentOffset + deltaX;
      if (newOffset > 0) newOffset = 0;
      if (newOffset < -40) newOffset = -40;
      panoBg.style.right = newOffset + '%';
      
      const calculatedAngle = Math.round(Math.abs(newOffset / 40) * 360);
      if (angleDisplay) {
        angleDisplay.textContent = calculatedAngle.toLocaleString('fa-IR') + '°';
      }
    };

    const handleTouchEnd = (e) => {
      if (!isDragging) return;
      isDragging = false;
      const computedRight = parseFloat(panoBg.style.right || '-20');
      currentOffset = isNaN(computedRight) ? -20 : computedRight;
    };

    container.addEventListener('touchstart', handleTouchStart, { passive: true });
    container.addEventListener('touchmove', handleTouchMove, { passive: true });
    container.addEventListener('touchend', handleTouchEnd);
    container.addEventListener('mousedown', handleTouchStart);
    window.addEventListener('mousemove', handleTouchMove);
    window.addEventListener('mouseup', handleTouchEnd);
  }

      if (btnVip) btnVip.className = 'veh-btn p-3 rounded-xl bg-surface text-on-surface text-right transition-all';
    }
    updateCustomizer();
  }

  function updateCustomizer() {
    const roomCheckbox = document.getElementById('room-upgrade');
    roomExtra = (roomCheckbox && roomCheckbox.checked) ? 1200000 : 0;

    const totalAddons = dietaryExtra + vehicleExtra + roomExtra;
    const finalTotal = BASE_PRICE + totalAddons;

    const addonPriceEl = document.getElementById('addon-price');
    const billTotalEl = document.getElementById('bill-total');
    const stickyTotalEl = document.getElementById('sticky-total');

    if (addonPriceEl) addonPriceEl.textContent = '+' + totalAddons.toLocaleString('fa-IR') + ' تومان';
    if (billTotalEl) billTotalEl.textContent = finalTotal.toLocaleString('fa-IR') + ' تومان';
    if (stickyTotalEl) stickyTotalEl.textContent = finalTotal.toLocaleString('fa-IR');
  }

  function shareTrip() {

        url: window.location.href,
      }).catch(() => {});
    } else {
      alert('لینک تور اختصاصی کپی شد.');
    }
  }
